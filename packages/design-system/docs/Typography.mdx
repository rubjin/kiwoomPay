import textStyles from '../tokens/figma/text-styles.json';
import roles from '../tokens/custom/typography-roles.json';

<Meta title="Foundation/Typography" />

export const kebab = (s) =>
  s
    .replace(/([a-z])([A-Z])/g, '$1-$2')
    .replace(/\s+/g, '-')
    .toLowerCase();

export const toMixinName = (figmaName) =>
  figmaName
    .split('/')
    .map((part) => kebab(part.trim()))
    .join('-');

export const roleNames = Object.entries(roles)
  .filter(([group]) => !group.startsWith('$'))
  .flatMap(([group, items]) =>
    Object.entries(items).map(([key, token]) => ({
      name: `${kebab(group)}-${kebab(key)}`,
      target: toMixinName(token.$value.slice(1, -1).replace(/\./g, '/')),
    })),
  );

export const styles = [...textStyles.textStyles].sort(
  (a, b) => b.fontSize - a.fontSize || b.fontWeight - a.fontWeight,
);

# Typography

글자는 Figma 텍스트 스타일(`tokens/figma/text-styles.json`)을 그대로 씁니다. 폰트는 Pretendard입니다.

## 1. 쓰는 법

컴포넌트 SCSS에는 **Figma 스타일 이름**을 적습니다. 이름에 크기와 굵기가 들어 있어 시안의 Inspect 값과 바로 대조할 수 있습니다.

```scss
// Figma "Typo / 16 / SemiBold"
.title {
  @include m.typography(typo-16-semi-bold);
}
```

- 규칙: `typo-<크기>-<regular | medium | semi-bold | bold>`
- 없는 이름을 쓰면 빌드 에러가 납니다(오타 방지).
- `font-size` · `font-weight`를 따로 덮어쓰지 않습니다. 굵기가 다르면 그 굵기의 스타일을 고릅니다
  (예: 16 SemiBold + `font-weight: 700` ❌ → `typo-16-bold` ✅).

## Do / Don't

- ✅ 시안 Inspect의 `Typo / 13 / Medium` → `typo-13-medium`
- ❌ 컴포넌트 전용 글자 이름을 새로 만들기(예: `badge-md`) — 이름만 보고 값을 알 수 없습니다
- ❌ mixin 뒤에 `font-size` · `font-weight`를 덮어쓰기
